cmake_minimum_required(VERSION 3.25)
project(os_subsystems VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build on Linux. On macOS, use make test (Docker).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SANITIZER "none" CACHE STRING "none, address, undefined, or thread")
set_property(CACHE SANITIZER PROPERTY STRINGS none address undefined thread)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -g -fno-omit-frame-pointer
  "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
if(NOT SANITIZER STREQUAL "none")
  if(NOT SANITIZER MATCHES "^(address|undefined|thread)$")
    message(FATAL_ERROR "Unsupported SANITIZER=${SANITIZER}")
  endif()
  add_compile_options(-fsanitize=${SANITIZER} -fno-sanitize-recover=all)
  add_link_options(-fsanitize=${SANITIZER} -fno-sanitize-recover=all)
endif()
find_package(Threads REQUIRED)
include(CTest)
add_library(os_common INTERFACE)
target_include_directories(os_common INTERFACE include)
target_link_libraries(os_common INTERFACE Threads::Threads)
add_executable(os_info src/info.cpp)
target_link_libraries(os_info PRIVATE os_common)
install(TARGETS os_info RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(common_test tests/common_test.cpp)
  target_link_libraries(common_test PRIVATE os_common)
  add_test(NAME common COMMAND common_test)
endif()
foreach(component rpc sched vm fs sync bench)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${component}/CMakeLists.txt")
    add_subdirectory(${component})
  endif()
endforeach()
