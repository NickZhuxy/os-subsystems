add_library(os_fs format.cpp image.cpp check.cpp)

target_link_libraries(os_fs PUBLIC os_common)
foreach(tool mkfs check cli)
  add_executable(fs_${tool} ${tool}_main.cpp)
  target_link_libraries(fs_${tool} PRIVATE os_fs)
  install(TARGETS fs_${tool} RUNTIME DESTINATION bin)
endforeach()

install(TARGETS fs_mount RUNTIME DESTINATION bin)
